  List<Element> select(dynamic args) {
    final all = tree();

    if (args['ancestor'] != null) {
      final parents = all.where((e) => matches(e.widget.key, args['ancestor'])).toList();
      if (parents.length != 1) throw parents.isEmpty ? 'ANCESTOR_NOT_FOUND' : 'AMBIGUOUS_ANCESTOR';
      return tree(parents.single).where((e) => matches(e.widget.key, args['key'])).toList();
    }
    return all.where((e) => matches(e.widget.key, args['key'])).toList();
  }
  bool interactive(Element e) {
    if (e.widget is EditableText || e.widget is Scrollable) return true;
    try { final dynamic w = e.widget; w.onPressed; return true; } catch (_) {}
    try { final dynamic w = e.widget; if (w.onTap != null || w.onLongPress != null || w.onPanUpdate != null) return true; } catch (_) {}
    return false;
  }
  bool enabled(Element e) {
    try { final dynamic w = e.widget; if (w.onPressed == null) return false; } catch (_) {}
    if (e.widget is EditableText) return !(e.widget as EditableText).readOnly;
    try { final dynamic w = e.widget; if (w.enabled == false) return false; } catch (_) {}
    return true;
  }
  Element? actionTarget(Element e) {
    if (interactive(e)) return e;
    final descendants = tree(e).where(interactive).toList();
    if (descendants.isEmpty) return null;
    final candidates = <Element>[];
    for (final child in descendants) {
      var nested = false;
      child.visitAncestorElements((parent) {
        if (identical(parent, e)) return false;
        if (interactive(parent)) { nested = true; return false; }
        return true;
      });
      if (!nested) candidates.add(child);
    }
    return candidates.length == 1 ? candidates.single : null;
  }

      if (current is RenderView) return current.flutterView.viewId;

  }

  }

  Future<Map<String, dynamic>> pointer(Element e, [dynamic scrollArgs]) async {
    final box = boxFor(e);
    if (!enabled(e)) return fail('NOT_ENABLED', '目标当前不可操作。');
    if (box == null) return fail('NOT_VISIBLE', '目标尚未布局或不可见。');
    final center = box.localToGlobal(box.size.center(Offset.zero));
    if (!hit(e, box, center)) return fail('NOT_HIT_TESTABLE', '目标被遮挡或不接收输入事件。');

      final extent = vertical ? box.size.height : box.size.width;
      final distance = requested < extent * .65 ? requested : extent * .65;
      final signed = direction == 'down' || direction == 'right' ? -distance : distance;
      delta = vertical ? Offset(0, signed) : Offset(signed, 0);
      start = center - delta / 2;

          if (state['active'] != true) return fail('SESSION_INVALIDATED', 'UI 会话已失效。');
          GestureBinding.instance.handlePointerEvent(PointerMoveEvent(pointer: id, position: start + delta * (step / 12), delta: delta / 12, viewId: viewId));
        }
      } else {

      }
      if (state['active'] != true) return fail('SESSION_INVALIDATED', 'UI 会话已失效。');
      GestureBinding.instance.handlePointerEvent(PointerUpEvent(pointer: id, position: start + delta, viewId: viewId));
      down = false;

        if (e.widget is Scrollable) {
          final scrollables = tree(parent).where((child) => child.widget is Scrollable).toList();
          if (scrollables.length == 1 && identical(scrollables.single, e)) yes = true;
        }
        return false;
      }
      return true;
    });
    return yes;
